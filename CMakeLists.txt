cmake_minimum_required(VERSION 3.18)
project(cardputer_lofi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LOFI_SANITIZERS "Enable host address/undefined behavior sanitizers" OFF)
if(LOFI_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS firmware/src/core/*.cpp firmware/src/audio/*.cpp firmware/src/ui/*.cpp)
add_library(lofi_core ${CORE_SOURCES})
target_include_directories(lofi_core PUBLIC firmware/include)
target_compile_options(lofi_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lofi_native native/main.cpp)
target_link_libraries(lofi_native PRIVATE lofi_core)
find_package(SDL2 CONFIG QUIET)
if(SDL2_FOUND)
  target_link_libraries(lofi_native PRIVATE SDL2::SDL2)
  target_compile_definitions(lofi_native PRIVATE LOFI_HAS_SDL=1)
else()
  message(STATUS "SDL2 unavailable: audio and screenshot export remain available")
endif()
enable_testing()
foreach(NAME music state controller storage ui)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${NAME}.cpp")
    add_executable(test_${NAME} tests/test_${NAME}.cpp)
    target_link_libraries(test_${NAME} PRIVATE lofi_core)
    target_compile_options(test_${NAME} PRIVATE -UNDEBUG)
    add_test(NAME ${NAME} COMMAND test_${NAME})
  endif()
endforeach()
