cmake_minimum_required(VERSION 3.18)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LOFI_VERSION LIMIT_COUNT 1)
if(NOT LOFI_VERSION MATCHES "^[0-9a-z.-]+$")
  message(FATAL_ERROR "Unsafe VERSION value")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
string(REGEX REPLACE "-.*$" "" LOFI_NUMERIC_VERSION "${LOFI_VERSION}")
project(cardputer_lofi VERSION ${LOFI_NUMERIC_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LOFI_SANITIZERS "Enable host address/undefined behavior sanitizers" OFF)
if(LOFI_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS firmware/src/core/*.cpp firmware/src/audio/*.cpp firmware/src/ui/*.cpp)
add_library(lofi_core ${CORE_SOURCES})
target_include_directories(lofi_core PUBLIC firmware/include)
target_compile_definitions(lofi_core PUBLIC LOFI_VERSION="${LOFI_VERSION}")
target_compile_options(lofi_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lofi_native native/main.cpp)
target_link_libraries(lofi_native PRIVATE lofi_core)
find_package(SDL2 CONFIG QUIET)
if(SDL2_FOUND)
  target_link_libraries(lofi_native PRIVATE SDL2::SDL2)
  target_compile_definitions(lofi_native PRIVATE LOFI_HAS_SDL=1)
else()
  message(STATUS "SDL2 unavailable: audio and screenshot export remain available")
endif()
enable_testing()
foreach(NAME music state controller storage ui)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${NAME}.cpp")
    add_executable(test_${NAME} tests/test_${NAME}.cpp)
    target_link_libraries(test_${NAME} PRIVATE lofi_core)
    target_compile_options(test_${NAME} PRIVATE -UNDEBUG)
    add_test(NAME ${NAME} COMMAND test_${NAME})
  endif()
endforeach()

# Keep ordinary offline/native builds independent of Arduino dependencies.
set(LOFI_M5GFX_SOURCE_DIR "" CACHE PATH "M5GFX source checkout for the LCD byte-order regression")
if(LOFI_M5GFX_SOURCE_DIR)
  if(NOT EXISTS "${LOFI_M5GFX_SOURCE_DIR}/src/lgfx/v1/misc/pixelcopy.hpp")
    message(FATAL_ERROR "LOFI_M5GFX_SOURCE_DIR must point to an M5GFX checkout")
  endif()
  add_executable(test_display_rgb565 tests/test_display_rgb565.cpp)
  target_link_libraries(test_display_rgb565 PRIVATE lofi_core)
  target_include_directories(test_display_rgb565 SYSTEM PRIVATE "${LOFI_M5GFX_SOURCE_DIR}/src")
  target_compile_options(test_display_rgb565 PRIVATE -UNDEBUG)
  add_test(NAME display_rgb565 COMMAND test_display_rgb565)
endif()
